// စာသားပြောင်းလဲတိုင်း တွက်ချက်မည့် Event scriptTextArea.addEventListener('input', () => { const text = scriptTextArea.value.trim(); const selectedLang = languageSelect.value; // User ရွေးထားတဲ့ ဘာသာစကား (ဥပမာ - 'MM', 'EN') let totalCount = 0; if (text === "") { totalCount = 0; } // ၁။ မြန်မာစာ (MM) ဖြစ်ခဲ့လျှင် - အက္ခရာ + ဗျည်းဗျဉ် (Characters) ကို အခြေခံပြီး တွက်ချက်ခြင်း else if (selectedLang === 'MM' || selectedLang === 'မြန်မာ') { // မြန်မာစာတွင် Space မခြားတတ်သဖြင့် စာလုံးရေ (Words) ထက် စာသားအရှည် (Length) ဖြင့် တွက်မှ တိကျပါမည် totalCount = text.length; } // ၂။ အင်္ဂလိပ်စာ (EN) ဖြစ်ခဲ့လျှင် - စာလုံးရေ (Words) ကို Space ဖြင့် ခွဲတွက်ခြင်း else if (selectedLang === 'EN' || selectedLang === 'English') { totalCount = text.split(/\s+/).length; } // ၃။ တခြား ဘာသာစကားများ (ဥပမာ - တရုတ်/ဂျပန် စာလုံးများ) အတွက် Character အလိုက်တွက်ခြင်း else { // Space ပါသော အနောက်တိုင်းဘာသာစကားဆိုလျှင် Words တွက်ပြီး၊ စာလုံးကပ်ရေးသော အရှေ့တိုင်းစာဆိုလျှင် Length တွက်ရန် totalCount = text.includes(' ') ? text.split(/\s+/).length : text.length; } // UI တွင် Dynamic ပြောင်းလဲပြသခြင်း wordCounter.textContent = `${totalCount} / 3000 words`; // ဥပမာ Milestone ဖြစ်တဲ့ ၁၅၀၀ ကျော်ရင် Warning ပြောင်းမယ် if (totalCount >= 1500 && totalCount < 3000) { wordCounter.style.color = '#eab308'; // ဝါဂွမ်းရောင် (Warning) } else if (totalCount >= 3000) { wordCounter.style.color = '#ef4444'; // အနီရောင် (Limit ပြည့်ပြီ) // Limit ထက်ကျော်သွားပါက ဖြတ်တောက်ရန် if (selectedLang === 'MM' || selectedLang === 'မြန်မာ') { scriptTextArea.value = text.substring(0, 3000); // မြန်မာစာကို Character ၃၀၀၀ တွင် ဖြတ်မည် } else { scriptTextArea.value = text.split(/\s+/).slice(0, 3000).join(" "); // English ကို စာလုံး ၃၀၀၀ တွင် ဖြတ်မည် } } else { wordCounter.style.color = ''; // ပုံမှန်အရောင် } });
